cmake_minimum_required(VERSION 3.20)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/instrumentation.patch" _bithac_patch_hash)
include(../common/patch-provenance.cmake)
turtle_check_patch_manifest(bithac "${_bithac_patch_hash}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/instrumentation.patch")
foreach(variant baseline cached)
  set(prefix "${TURTLE_DEPS_ROOT}/install/bithac-${variant}")
  if(NOT EXISTS "${prefix}/lib/libwolfssl.a")
    message(FATAL_ERROR "Build the author implementation pair with scripts/build-bithac.sh first")
  endif()
  set(target turtle_adapter_bithac_${variant})
  if(variant STREQUAL "cached")
    set(adapter_id bithac)
    set(enabled 1)
  else()
    set(adapter_id bithac-baseline)
    set(enabled 0)
  endif()
  add_library(${target} MODULE ../wolfssl/wolfssl_adapter.cpp)
  target_compile_definitions(${target} PRIVATE TURTLE_LIBRARY_LINKAGE="static")
  set_property(TARGET ${target} PROPERTY TURTLE_LIBRARY_LINKAGE "static")
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/include")
  target_include_directories(${target} SYSTEM PRIVATE "${prefix}/include")
  target_link_libraries(${target} PRIVATE "${prefix}/lib/libwolfssl.a" m pthread)
  target_compile_definitions(${target} PRIVATE TURTLE_BITHAC=1
    TURTLE_BITHAC_ENABLED=${enabled} TURTLE_ADAPTER_ID="${adapter_id}"
    TURTLE_PATCH_SHA256="${_bithac_patch_hash}" HAVE_SECRET_CALLBACK SHOW_SECRETS
    WOLFSSL_TLS13_MIDDLEBOX_COMPAT WOLFSSL_GETRANDOM)
  if(enabled)
    target_compile_definitions(${target} PRIVATE WOLFSSL_CACHED_INFO WOLFSSL_SSC)
  endif()
  set_target_properties(${target} PROPERTIES PREFIX "" OUTPUT_NAME "${adapter_id}"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/adapters")
endforeach()
