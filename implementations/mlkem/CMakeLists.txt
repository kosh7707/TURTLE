cmake_minimum_required(VERSION 3.20)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/instrumentation.patch" _mlkem_patch_hash)
include(../common/patch-provenance.cmake)
turtle_check_patch_manifest(mlkem "${_mlkem_patch_hash}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/instrumentation.patch")
foreach(variant c avx2)
  set(prefix "${TURTLE_DEPS_ROOT}/install/mlkem-${variant}")
  if(NOT EXISTS "${prefix}/lib/libwolfssl.a")
    message(FATAL_ERROR "Build the ML-KEM comparison pair with scripts/build-mlkem.sh first")
  endif()
  set(target turtle_adapter_mlkem_${variant})
  set(enabled 0)
  if(variant STREQUAL "avx2")
    set(enabled 1)
  endif()
  add_library(${target} MODULE ../wolfssl/wolfssl_adapter.cpp)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/include")
  target_include_directories(${target} SYSTEM PRIVATE "${prefix}/include")
  target_link_libraries(${target} PRIVATE "${prefix}/lib/libwolfssl.a" m pthread)
  target_compile_definitions(${target} PRIVATE TURTLE_MLKEM=1 TURTLE_MLKEM_AVX2=${enabled}
    TURTLE_ADAPTER_ID="mlkem-${variant}" TURTLE_PATCH_SHA256="${_mlkem_patch_hash}"
    HAVE_SECRET_CALLBACK SHOW_SECRETS WOLFSSL_TLS13_MIDDLEBOX_COMPAT WOLFSSL_GETRANDOM)
  set_target_properties(${target} PROPERTIES PREFIX "" OUTPUT_NAME "mlkem-${variant}"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/adapters")
endforeach()
