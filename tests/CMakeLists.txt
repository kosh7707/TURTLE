cmake_minimum_required(VERSION 3.20)
project(turtle_message_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
set(TURTLE_SOURCE_BUILD "${ROOT}/build-message" CACHE PATH "Separate source build")
set(TURTLE_PKI "${ROOT}/pki/default" CACHE PATH "Local test certificates")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_library(contract_c OBJECT contract.c)
target_include_directories(contract_c PRIVATE "${ROOT}/include")
add_executable(core_test core.cpp)
target_include_directories(core_test PRIVATE "${ROOT}/include" "${ROOT}/src")
target_link_libraries(core_test PRIVATE "${TURTLE_SOURCE_BUILD}/libturtle_core.a" pthread)
target_link_options(core_test PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
target_compile_options(core_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME deterministic-core COMMAND core_test)
add_test(NAME report-contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/report_test.py")
foreach(suite matrix research external)
  add_test(NAME tls-${suite} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/tls_test.py" ${suite} --build "${TURTLE_SOURCE_BUILD}" --pki "${TURTLE_PKI}" --output "${CMAKE_CURRENT_BINARY_DIR}/evidence")
  set_tests_properties(tls-${suite} PROPERTIES TIMEOUT 180)
endforeach()
add_test(NAME lookup-observations COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/observation_test.py" --build "${TURTLE_SOURCE_BUILD}" --pki "${TURTLE_PKI}" --output "${CMAKE_CURRENT_BINARY_DIR}/evidence")
set_tests_properties(lookup-observations PROPERTIES TIMEOUT 30)
