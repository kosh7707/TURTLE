cmake_minimum_required(VERSION 3.20)
project(turtle_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# The production build does not depend on the tests.
set(TURTLE_ADAPTER_LINKAGE module CACHE STRING "" FORCE)
set(TURTLE_BUILD_LOOKUP ON CACHE BOOL "" FORCE)
set(TURTLE_BUILD_MODULES ON CACHE BOOL "" FORCE)
set(TURTLE_WITH_OPENSSL ON CACHE BOOL "" FORCE)
add_subdirectory(.. framework)
enable_testing()
# Several small native tests use assert, including expressions that exercise
# the API. Keep assertions active even when testing a Release production build.
add_compile_options(-Wall -Wextra -Wpedantic -UNDEBUG)
if(TURTLE_WARNINGS_AS_ERRORS)
  add_compile_options(-Werror)
endif()
foreach(module config control transport measure capability record lookup lifecycle report adapters)
  add_subdirectory(${module})
endforeach()
if(TARGET turtle_interpret AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/interpret/CMakeLists.txt")
  add_subdirectory(interpret)
endif()
add_subdirectory(integration)

foreach(name openssl wolfssl mbedtls)
  if(TARGET turtle_adapter_${name})
    add_test(NAME read_pattern_${name} COMMAND turtle_adapter_read_pattern
      "$<TARGET_FILE:turtle_adapter_${name}>" "${CMAKE_BINARY_DIR}/pki")
    set_tests_properties(read_pattern_${name} PROPERTIES
      FIXTURES_REQUIRED local_pki TIMEOUT 30 LABELS integration)
  endif()
endforeach()
