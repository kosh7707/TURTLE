cmake_minimum_required(VERSION 3.20)
project(turtle VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(TURTLE_WARNINGS_AS_ERRORS "Treat TURTLE compiler warnings as errors" OFF)
option(TURTLE_WITH_OPENSSL "Build the OpenSSL adapter" ON)
option(TURTLE_WITH_WOLFSSL "Build the pinned wolfSSL adapter" OFF)
option(TURTLE_WITH_MBEDTLS "Build the pinned Mbed TLS adapter" OFF)
option(TURTLE_WITH_INTERPRETER "Build the offline TLS interpreter" ON)
set(TURTLE_DEPS_ROOT "$ENV{TURTLE_DEPS_ROOT}" CACHE PATH "Locally built TLS dependencies")
if(NOT TURTLE_DEPS_ROOT)
  set(TURTLE_DEPS_ROOT "${PROJECT_SOURCE_DIR}/.deps")
endif()
if(EXISTS "${TURTLE_DEPS_ROOT}/openssl/include/openssl/ssl.h")
  set(OPENSSL_ROOT_DIR "${TURTLE_DEPS_ROOT}/openssl" CACHE PATH "OpenSSL installation")
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
if(TURTLE_WARNINGS_AS_ERRORS)
  add_compile_options(-Werror)
endif()

add_library(turtle_model src/model/model.cpp)
target_include_directories(turtle_model PUBLIC src include)
string(TOUPPER "${CMAKE_BUILD_TYPE}" build_type_upper)
set(TURTLE_SOURCE_REVISION "unavailable")
find_package(Git QUIET)
if(Git_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" describe --always --dirty
    WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
    OUTPUT_VARIABLE TURTLE_SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET)
endif()
set(TURTLE_BUILD_INFORMATION
  "compiler=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}; path=${CMAKE_CXX_COMPILER}; type=${CMAKE_BUILD_TYPE}; flags=${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${build_type_upper}}; standard=C++20; warnings=-Wall -Wextra -Wpedantic; werror=${TURTLE_WARNINGS_AS_ERRORS}; source=${TURTLE_SOURCE_REVISION}")
configure_file(src/model/build_info.hpp.in generated/model/build_info.hpp @ONLY)
file(WRITE "${CMAKE_BINARY_DIR}/build-manifest.txt" "${TURTLE_BUILD_INFORMATION}\n")
target_include_directories(turtle_model PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_library(turtle_config src/config/config.cpp)
target_link_libraries(turtle_config PUBLIC turtle_model)
target_include_directories(turtle_config SYSTEM PRIVATE third_party)
foreach(module control transport measure capability record lookup lifecycle report)
  add_subdirectory(src/${module})
endforeach()
if(TURTLE_WITH_INTERPRETER)
  add_subdirectory(src/interpret)
endif()
add_executable(turtle src/main.cpp)
target_link_libraries(turtle PRIVATE turtle_config turtle_lifecycle)
set_target_properties(turtle PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(tool turtle-report turtle-interpret)
  if(TARGET ${tool})
    set_target_properties(${tool} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  endif()
endforeach()

# Adapters see only the public implementation contract, not src/.
add_subdirectory(implementations/mock)
if(TURTLE_WITH_OPENSSL)
  add_subdirectory(implementations/openssl)
endif()
if(TURTLE_WITH_WOLFSSL)
  add_subdirectory(implementations/wolfssl)
endif()
if(TURTLE_WITH_MBEDTLS)
  add_subdirectory(implementations/mbedtls)
endif()
