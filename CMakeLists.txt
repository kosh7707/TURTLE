cmake_minimum_required(VERSION 3.20)
project(turtle VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(TURTLE_WARNINGS_AS_ERRORS "Treat TURTLE compiler warnings as errors" OFF)
option(TURTLE_BUILD_CLIENT "Build the independent client" ON)
option(TURTLE_BUILD_SERVER "Build the independent server" ON)
option(TURTLE_BUILD_LOOKUP "Build the independent optional lookup service" OFF)
option(TURTLE_BUILD_ANALYSIS "Build offline report and interpreter" ON)
set(TURTLE_ADAPTER_LINKAGE "builtin" CACHE STRING "Adapter connection: builtin or module")
set_property(CACHE TURTLE_ADAPTER_LINKAGE PROPERTY STRINGS builtin module)
set(TURTLE_IMPLEMENTATION "openssl" CACHE STRING "One builtin adapter implementation")
option(TURTLE_BUILD_MODULES "Build optional adapter modules for experiments" OFF)
option(TURTLE_WITH_OPENSSL "Build the optional OpenSSL module" OFF)
option(TURTLE_WITH_WOLFSSL "Build the optional wolfSSL module" OFF)
option(TURTLE_WITH_MBEDTLS "Build the optional Mbed TLS module" OFF)
option(TURTLE_WITH_BITHAC "Build the optional BiTHaC pair" OFF)
option(TURTLE_WITH_MLKEM "Build the optional ML-KEM pair" OFF)
option(TURTLE_WITH_INTERPRETER "Build offline TLS interpreter with analysis" ON)
if(NOT TURTLE_ADAPTER_LINKAGE MATCHES "^(builtin|module)$")
  message(FATAL_ERROR "TURTLE_ADAPTER_LINKAGE must be builtin or module")
endif()
set(TURTLE_DEPS_ROOT "$ENV{TURTLE_DEPS_ROOT}" CACHE PATH "Locally built TLS dependencies")
if(NOT TURTLE_DEPS_ROOT)
  set(TURTLE_DEPS_ROOT "${PROJECT_SOURCE_DIR}/.deps")
endif()
if(EXISTS "${TURTLE_DEPS_ROOT}/openssl/include/openssl/ssl.h")
  set(OPENSSL_ROOT_DIR "${TURTLE_DEPS_ROOT}/openssl" CACHE PATH "OpenSSL installation")
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
if(TURTLE_WARNINGS_AS_ERRORS)
  add_compile_options(-Werror)
endif()

add_library(turtle_model src/model/model.cpp)
target_include_directories(turtle_model PUBLIC src include)
string(TOUPPER "${CMAKE_BUILD_TYPE}" build_type_upper)
set(TURTLE_SOURCE_REVISION "unavailable")
find_package(Git QUIET)
if(Git_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" describe --always --dirty
    WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
    RESULT_VARIABLE _turtle_git_status
    OUTPUT_VARIABLE _turtle_git_revision OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET)
  if(_turtle_git_status EQUAL 0)
    set(TURTLE_SOURCE_REVISION "${_turtle_git_revision}")
  endif()
endif()
set(TURTLE_BUILD_INFORMATION
  "compiler=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}; path=${CMAKE_CXX_COMPILER}; type=${CMAKE_BUILD_TYPE}; flags=${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${build_type_upper}}; standard=C++20; warnings=-Wall -Wextra -Wpedantic; werror=${TURTLE_WARNINGS_AS_ERRORS}; source=${TURTLE_SOURCE_REVISION}; target=${CMAKE_SYSTEM_NAME}/${CMAKE_SYSTEM_PROCESSOR}; cross=${CMAKE_CROSSCOMPILING}; adapter=${TURTLE_ADAPTER_LINKAGE}/${TURTLE_IMPLEMENTATION}")
configure_file(src/model/build_info.hpp.in generated/model/build_info.hpp @ONLY)
file(WRITE "${CMAKE_BINARY_DIR}/build-manifest.txt" "${TURTLE_BUILD_INFORMATION}\n")
target_include_directories(turtle_model PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(TURTLE_BUILD_CLIENT OR TURTLE_BUILD_SERVER OR TURTLE_BUILD_LOOKUP)
  add_library(turtle_config src/config/config.cpp)
  target_link_libraries(turtle_config PUBLIC turtle_model)
  target_include_directories(turtle_config SYSTEM PRIVATE third_party)
  foreach(module control transport measure capability record lookup lifecycle)
    add_subdirectory(src/${module})
  endforeach()
endif()
if(TURTLE_BUILD_ANALYSIS)
  add_subdirectory(src/report)
  if(TURTLE_WITH_INTERPRETER)
    add_subdirectory(src/interpret)
  endif()
endif()
foreach(tool turtle-report turtle-interpret)
  if(TARGET ${tool})
    set_target_properties(${tool} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  endif()
endforeach()

# Dependency discovery is conditional on the selected endpoint implementation.
# No target program is run by configure (including cross compilation).
if(TURTLE_BUILD_CLIENT OR TURTLE_BUILD_SERVER)
  if(TURTLE_ADAPTER_LINKAGE STREQUAL "builtin")
    if(TURTLE_IMPLEMENTATION MATCHES "^bithac")
      set(TURTLE_WITH_BITHAC ON)
    elseif(TURTLE_IMPLEMENTATION MATCHES "^mlkem-")
      set(TURTLE_WITH_MLKEM ON)
    else()
      string(TOUPPER "${TURTLE_IMPLEMENTATION}" selected_upper)
      set(TURTLE_WITH_${selected_upper} ON)
    endif()
  endif()
  foreach(adapter openssl wolfssl mbedtls bithac mlkem)
    string(TOUPPER "${adapter}" adapter_upper)
    if(TURTLE_WITH_${adapter_upper})
      add_subdirectory(implementations/${adapter})
    endif()
  endforeach()
  if(TURTLE_BUILD_MODULES)
    add_subdirectory(implementations/mock)
  endif()
  if(TURTLE_ADAPTER_LINKAGE STREQUAL "builtin")
    string(REPLACE "-" "_" selected "${TURTLE_IMPLEMENTATION}")
    if(selected STREQUAL "bithac")
      set(selected bithac_cached)
    endif()
    set(adapter_target turtle_adapter_${selected})
    if(NOT TARGET ${adapter_target})
      message(FATAL_ERROR "Unknown builtin implementation ${TURTLE_IMPLEMENTATION}")
    endif()
    get_target_property(adapter_sources ${adapter_target} SOURCES)
    get_target_property(adapter_source_dir ${adapter_target} SOURCE_DIR)
    set(builtin_sources)
    foreach(source IN LISTS adapter_sources)
      list(APPEND builtin_sources "${adapter_source_dir}/${source}")
    endforeach()
    add_library(turtle_builtin STATIC ${builtin_sources})
    foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS)
      get_target_property(value ${adapter_target} ${property})
      if(value)
        set_property(TARGET turtle_builtin PROPERTY ${property} "${value}")
      endif()
    endforeach()
    # Preserve the actual TLS library link inputs independently of adapter mode.
    get_target_property(tls_inputs ${adapter_target} LINK_LIBRARIES)
    target_link_libraries(turtle_builtin PUBLIC ${tls_inputs})
    get_target_property(tls_linkage ${adapter_target} TURTLE_LIBRARY_LINKAGE)
  else()
    set(tls_linkage "module-owned; see adapter build options and module dependencies")
  endif()
  if(NOT TURTLE_BUILD_MODULES)
    foreach(adapter openssl wolfssl mbedtls bithac_baseline bithac_cached mlkem_c mlkem_avx2)
      if(TARGET turtle_adapter_${adapter})
        set_target_properties(turtle_adapter_${adapter} PROPERTIES EXCLUDE_FROM_ALL TRUE)
      endif()
    endforeach()
  endif()
  string(REPLACE ";" "," tls_linkage "${tls_linkage}")
  foreach(role client server)
    string(TOUPPER "${role}" role_upper)
    if(TURTLE_BUILD_${role_upper})
      add_executable(turtle-${role} src/main.cpp)
      target_link_libraries(turtle-${role} PRIVATE turtle_config turtle_lifecycle)
      if(role STREQUAL "client")
        set(role_enum Client)
      else()
        set(role_enum Server)
      endif()
      target_compile_definitions(turtle-${role} PRIVATE
        TURTLE_ENDPOINT_ROLE=turtle::Role::${role_enum}
        TURTLE_ADAPTER_LINKAGE="${TURTLE_ADAPTER_LINKAGE}"
        TURTLE_IMPLEMENTATION="${TURTLE_IMPLEMENTATION}"
        TURTLE_TLS_LINKAGE="${tls_linkage}")
      if(TURTLE_ADAPTER_LINKAGE STREQUAL "builtin")
        target_link_libraries(turtle-${role} PRIVATE turtle_builtin)
        target_compile_definitions(turtle-${role} PRIVATE TURTLE_BUILTIN_ADAPTER=1)
      endif()
      set_target_properties(turtle-${role} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
      install(TARGETS turtle-${role} RUNTIME DESTINATION bin)
    endif()
  endforeach()
endif()

if(TURTLE_BUILD_CLIENT AND TURTLE_BUILD_SERVER)
  configure_file(scripts/run-local.py "${CMAKE_BINARY_DIR}/turtle-local" COPYONLY)
  configure_file(scripts/profile_config.py "${CMAKE_BINARY_DIR}/profile_config.py" COPYONLY)
endif()

if(TURTLE_BUILD_LOOKUP)
  add_executable(turtle-lookup src/lifecycle/lookup_main.cpp)
  target_link_libraries(turtle-lookup PRIVATE turtle_config turtle_lifecycle)
  set_target_properties(turtle-lookup PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endif()
