cmake_minimum_required(VERSION 3.20)
project(turtle LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(TURTLE_DEPS_ROOT "${CMAKE_SOURCE_DIR}/.deps-message" CACHE PATH "Pinned offline dependency cache")
set(TURTLE_IMPLEMENTATIONS "openssl;mbedtls;wolfssl;mlkem;bithac;openssl-padded" CACHE STRING "Source implementations to compile")
option(TURTLE_WARNINGS_AS_ERRORS "Warnings are errors" ON)
add_library(turtle_core STATIC src/allocation/allocation.cpp src/digest/digest.cpp src/mockca/mockca.cpp src/spec/spec.cpp src/observation/observation.cpp src/socket/socket.cpp src/execution/execution.cpp src/journal/journal.cpp)
target_include_directories(turtle_core PUBLIC include src)
target_include_directories(turtle_core SYSTEM PUBLIC third_party)
target_compile_options(turtle_core PRIVATE -Wall -Wextra -Wpedantic)
if(TURTLE_WARNINGS_AS_ERRORS)
  target_compile_options(turtle_core PRIVATE -Werror)
endif()
include(cmake/Implementation.cmake)
foreach(implementation IN LISTS TURTLE_IMPLEMENTATIONS)
  add_subdirectory(implementations/${implementation})
endforeach()

target_link_options(turtle_core INTERFACE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
